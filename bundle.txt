package ru.qa.course.core;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import java.util.zip.ZipEntry;
import java.util.zip.ZipFile;
/**
 * Подбор chromedriver под установленную версию Chrome.
 *
 * Порядок работы (выполняется один раз за прогон тестов):
 * 1. Определить версию Chrome: реестр Windows (BLBeacon), exe на macOS,
 *    либо переменная окружения CHROME_VERSION (можно подставить явно).
 * 2. Если вложенный drivers/chromedriver(.exe) совпадает по мажорной версии —
 *    использовать его.
 * 3. Иначе скачать подходящий драйвер через зеркало npmmirror
 *    (доступно из корпоративных сетей, где googlechromelabs заблокирован)
 *    во временную папку и подменить вложенный драйвер.
 * 4. Найденный путь выставляется в системное свойство
 *    webdriver.chrome.driver — Selenium Manager не обращается к сети.
 *
 * Если ничего не удалось — оставляем вложенный драйвер: Selenium выдаст
 * понятную ошибку несовпадения версий.
 */
public final class ChromeDriverResolver {

    private static final Path DRIVER_DIR;
    private static final Path BUNDLED_DRIVER;
    private static final String NPMMIRROR =
            "https://registry.npmmirror.com/-/binary/chrome-for-testing";

    static {
        // drivers/ лежит в корне проекта (pom.xml передаёт путь через свойство)
        String configured = System.getProperty("project.basedir", ".");
        DRIVER_DIR = Paths.get(configured, "drivers");
        BUNDLED_DRIVER = DRIVER_DIR.resolve(isWindows()
                ? "chromedriver.exe" : "chromedriver");
    }

    private ChromeDriverResolver() {
    }

    private static boolean isWindows() {
        return System.getProperty("os.name", "").toLowerCase().contains("win");
    }

    private static boolean isMac() {
        return System.getProperty("os.name", "").toLowerCase().contains("mac");
    }

    /** Версия Chrome: переменная окружения CHROME_VERSION имеет приоритет. */
    private static String chromeVersion() {
        String env = System.getenv("CHROME_VERSION");
        if (env != null && !env.isBlank()) {
            return env.trim();
        }
        if (isWindows()) {
            return chromeVersionWindows();
        }
        if (isMac()) {
            return chromeVersionMac();
        }
        return null;
    }

    /** Версия Chrome из реестра Windows. */
    private static String chromeVersionWindows() {
        String[] keys = {
                "HKCU\\Software\\Google\\Chrome\\BLBeacon",
                "HKLM\\Software\\Google\\Chrome\\BLBeacon"
        };
        for (String key : keys) {
            try {
                Process p = new ProcessBuilder("reg", "query", key, "/v", "version")
                        .start();
                String out = new String(p.getInputStream().readAllBytes(),
                        StandardCharsets.UTF_8);
                p.waitFor();
                Matcher m = Pattern.compile("(\\d+\\.\\d+\\.\\d+\\.\\d+)").matcher(out);
                if (m.find()) {
                    return m.group(1);
                }
            } catch (Exception ignored) {
                // пробуем следующий источник
            }
        }
        return null;
    }

    /** Версия Chrome на macOS. */
    private static String chromeVersionMac() {
        Path exe = Paths.get("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome");
        if (Files.exists(exe)) {
            try {
                Process p = new ProcessBuilder(exe.toString(), "--version").start();
                String out = new String(p.getInputStream().readAllBytes(),
                        StandardCharsets.UTF_8).trim();
                if (out.contains(" ")) {
                    return out.substring(out.lastIndexOf(' ') + 1);
                }
            } catch (Exception ignored) {
                // не удалось — вернём null
            }
        }
        return null;
    }

    /** Мажорная версия вложенного драйвера (или null, если файла нет). */
    private static String bundledMajor() {
        if (!Files.exists(BUNDLED_DRIVER)) {
            return null;
        }
        try {
            Process p = new ProcessBuilder(BUNDLED_DRIVER.toString(), "--version").start();
            String out = new String(p.getInputStream().readAllBytes(),
                    StandardCharsets.UTF_8);
            Matcher m = Pattern.compile("(\\d+)\\.\\d+\\.\\d+\\.\\d+").matcher(out);
            return m.find() ? m.group(1) : null;
        } catch (Exception e) {
            return null;
        }
    }

    /** Скачивает драйвер нужной мажорной версии через npmmirror. */
    private static Path downloadDriver(String major) {
        try {
            String listing = httpGet(NPMMIRROR);
            List<String> versions = new ArrayList<>();
            Matcher m = Pattern.compile("chrome-for-testing/(\\d+\\.\\d+\\.\\d+\\.\\d+)/")
                    .matcher(listing);
            while (m.find()) {
                String v = m.group(1);
                if (v.split("\\.")[0].equals(major)) {
                    versions.add(v);
                }
            }
            if (versions.isEmpty()) {
                System.err.println("[chromedriver] нет сборки для версии " + major);
                return null;
            }
            versions.sort(Comparator.comparing((String v) ->
                    Long.parseLong(v.split("\\.")[3])));
            String chosen = versions.get(versions.size() - 1);

            String platform = isWindows() ? "win64" : (isMac() ? "mac-arm64" : "linux64");
            String fileName = "chromedriver-" + platform + ".zip";
            String url = NPMMIRROR + "/" + chosen + "/" + fileName;
            System.err.println("[chromedriver] скачиваю " + chosen + " ...");
            byte[] data = httpGetBytes(url);

            Path zip = Files.createTempFile("chromedriver-", ".zip");
            Files.write(zip, data);
            Path target = DRIVER_DIR.resolve(isWindows()
                    ? "chromedriver.exe" : "chromedriver");
            Files.createDirectories(DRIVER_DIR);
            try (ZipFile zf = new ZipFile(zip.toFile())) {
                ZipEntry entry = zf.stream()
                        .filter(e -> e.getName().endsWith(
                                isWindows() ? "chromedriver.exe" : "chromedriver"))
                        .findFirst()
                        .orElseThrow(() -> new IllegalStateException(
                                "chromedriver не найден в архиве"));
                Files.copy(zf.getInputStream(entry), target,
                        StandardCopyOption.REPLACE_EXISTING);
            }
            Files.deleteIfExists(zip);
            if (!isWindows()) {
                try {
                    new ProcessBuilder("chmod", "+x", target.toString()).start().waitFor();
                } catch (Exception ignored) {
                    // не критично
                }
            }
            System.err.println("[chromedriver] установлен " + chosen);
            return target;
        } catch (Exception e) {
            System.err.println("[chromedriver] скачивание не удалось: " + e);
            return null;
        }
    }

    private static String httpGet(String url) {
        try {
            return new String(httpGetBytes(url), StandardCharsets.UTF_8);
        } catch (Exception e) {
            throw new IllegalStateException("HTTP GET failed: " + url, e);
        }
    }

    private static byte[] httpGetBytes(String url)
            throws IOException, InterruptedException {
        java.net.http.HttpClient client = java.net.http.HttpClient.newBuilder()
                .followRedirects(java.net.http.HttpClient.Redirect.ALWAYS).build();
        java.net.http.HttpRequest request = java.net.http.HttpRequest.newBuilder()
                .uri(java.net.URI.create(url))
                .header("User-Agent", "Mozilla/5.0")
                .GET().build();
        return client.send(request, java.net.http.HttpResponse.BodyHandlers.ofByteArray())
                .body();
    }

    /**
     * Возвращает путь к подходящему chromedriver и выставляет системное
     * свойство webdriver.chrome.driver. Вызывать один раз до создания драйвера.
     */
    public static String resolve() {
        Path resolved;
        String version = chromeVersion();
        String bundledMajor = bundledMajor();

        if (version == null) {
            System.err.println("[chromedriver] версия Chrome не определена, "
                    + "использую вложенный драйвер");
            resolved = BUNDLED_DRIVER;
        } else {
            String major = version.split("\\.")[0];
            if (major.equals(bundledMajor)) {
                resolved = BUNDLED_DRIVER;
            } else {
                System.err.println("[chromedriver] Chrome " + version
                        + ", вложенный драйвер " + bundledMajor + " — подбираю нужный");
                Path downloaded = downloadDriver(major);
                resolved = downloaded != null ? downloaded : BUNDLED_DRIVER;
            }
        }

        if (resolved != null && Files.exists(resolved)) {
            System.setProperty("webdriver.chrome.driver", resolved.toString());
            return resolved.toString();
        }
        return null;
    }
}

<script>
// Книги каталога
const BOOKS = [
  { id: 1, title: "Чистый код", price: 1290 },
  { id: 2, title: "Совершенный код", price: 1850 },
  { id: 3, title: "Тестирование Дот Ком", price: 990 },
  { id: 4, title: "Рефакторинг", price: 2100 }
];

const CREDENTIALS = { email: "qa@bookmarket.local", password: "correct-horse" };

let cart = {};  // id -> qty

const $ = (id) => document.getElementById(id);

function show(sectionId) {
  ["section-login", "section-catalog", "section-cart"].forEach(id => {
    $(id).classList.toggle("hidden", id !== sectionId);
  });
}

// Навигация зависит от состояния входа: до входа видна только кнопка «Вход»
function setNavVisibility(loggedIn) {
  $("nav-login").classList.toggle("hidden", loggedIn);
  $("nav-catalog").classList.toggle("hidden", !loggedIn);
  $("nav-cart").classList.toggle("hidden", !loggedIn);
}

function renderBadge() {
  const n = Object.values(cart).reduce((a, b) => a + b, 0);
  $("cart-badge").textContent = String(n);
}

function renderCart() {
  const box = $("cart-lines");
  box.innerHTML = "";
  let total = 0;
  for (const book of BOOKS) {
    if (!cart[book.id]) continue;
    total += book.price * cart[book.id];
    const line = document.createElement("div");
    line.className = "cart-line";
    line.setAttribute("data-test", "cart-line");
    line.innerHTML = "<span>" + book.title + " x" + cart[book.id] + "</span><span>" +
      (book.price * cart[book.id]) + " руб.</span>";
    box.appendChild(line);
  }
  if (total === 0) {
    box.innerHTML = "<div class='muted'>Корзина пуста</div>";
  }
  $("cart-total").textContent = String(total);
}

function renderCatalog() {
  const box = $("book-list");
  box.innerHTML = "";
  for (const book of BOOKS) {
    const row = document.createElement("div");
    row.className = "book-row";
    row.setAttribute("data-test", "book-" + book.id);
    row.innerHTML = "<span class='t'>" + book.title + "</span>" +
      "<span class='p'>" + book.price + " руб.</span>";
    const btn = document.createElement("button");
    btn.textContent = "В корзину";
    btn.setAttribute("data-test", "add-" + book.id);
    btn.addEventListener("click", () => {
      cart[book.id] = (cart[book.id] || 0) + 1;
      renderBadge();
      renderCart();
    });
    row.appendChild(btn);
    box.appendChild(row);
  }
}

// Вход: проверка кредов; ошибка показывается в #login-error
$("login-button").addEventListener("click", () => {
  const email = $("login-email").value.trim();
  const password = $("login-password").value;
  if (email === CREDENTIALS.email && password === CREDENTIALS.password) {
    $("login-error").textContent = "";
    sessionStorage.setItem("bm-logged-in", "1");
    renderCatalog();
    renderBadge();
    setNavVisibility(true);
    show("section-catalog");
  } else {
    $("login-error").textContent = "Неверный email или пароль";
  }
});

$("nav-login").addEventListener("click", () => show("section-login"));
$("nav-catalog").addEventListener("click", () => { renderCatalog(); show("section-catalog"); });
$("nav-cart").addEventListener("click", () => { renderCart(); show("section-cart"); });

// Если сессия уже была "залогинена" (тот же таб, перезагрузки страницы)
if (sessionStorage.getItem("bm-logged-in") === "1") {
  renderCatalog();
  setNavVisibility(true);
  show("section-catalog");
} else {
  setNavVisibility(false);
}
</script>
</body>
</html>

const BOOKS = [
  { id: 1, title: "Чистый код", price: 1290 },
  { id: 2, title: "Совершенный код", price: 1850 },

// Диапазон задержек: 0-499 мс. Иногда элемент готов мгновенно,
// иногда появляется через полсекунды — тесты без ожиданий
// падают непредсказуемо. Диапазон можно менять (материал для занятия).
const DELAY = () => Math.floor(Math.random() * 500);

let cart = {};

function show(sectionId) {
  ["section-login", "section-catalog", "section-cart"].forEach(id => {
    $(id).classList.toggle("hidden", id !== sectionId);
  });

  $("nav-login").classList.toggle("hidden", loggedIn);
  $("nav-catalog").classList.toggle("hidden", !loggedIn);
  $("nav-cart").classList.toggle("hidden", !loggedIn);

function renderBadge() {
  setTimeout(() => {
    const n = Object.values(cart).reduce((a, b) => a + b, 0);
    $("cart-badge").textContent = String(n);
  }, DELAY());

function renderCart() {
  setTimeout(() => {
    $("cart-spinner").classList.remove("hidden");
    const box = $("cart-lines");
    box.innerHTML = "";
    let total = 0;
    for (const book of BOOKS) {
      if (!cart[book.id]) continue;
      total += book.price * cart[book.id];
      const line = document.createElement("div");
      line.className = "cart-line";
      line.setAttribute("data-test", "cart-line");
      line.innerHTML = "<span>" + book.title + " x" + cart[book.id] + "</span><span>" +
        (book.price * cart[book.id]) + " руб.</span>";
      box.appendChild(line);
    }
    if (total === 0) {
      box.innerHTML = "<div class='muted'>Корзина пуста</div>";
    }
    $("cart-total").textContent = String(total);
    $("cart-spinner").classList.add("hidden");
  }, DELAY());
}

function renderCatalog() {
  setTimeout(() => {
    $("catalog-spinner").classList.remove("hidden");
    const box = $("book-list");
    box.innerHTML = "";
    // порядок книг перемешивается при каждой отрисовке —
    // локаторы по позиции нестабильны

    for (const book of shuffled) {
      const row = document.createElement("div");
      row.className = "book-row";
      row.setAttribute("data-test", "book-" + book.id);
      row.innerHTML = "<span class='t'>" + book.title + "</span>" +
        "<span class='p'>" + book.price + " руб.</span>";

      btn.textContent = "В корзину";
      btn.setAttribute("data-test", "add-" + book.id);
      btn.addEventListener("click", () => {
        cart[book.id] = (cart[book.id] || 0) + 1;
        renderBadge();
        renderCart();
      });
      row.appendChild(btn);
      box.appendChild(row);
    }
    $("catalog-spinner").classList.add("hidden");
  }, DELAY());
}

/**

    protected WebDriver driver;

    /**

    @RegisterExtension

            byte[] png = ((TakesScreenshot) driver).getScreenshotAs(OutputType.BYTES);
            Allure.addAttachment("Скриншот при падении", "image/png",
                    new ByteArrayInputStream(png), "png");

    void setUpBrowser() {
        // подбор chromedriver под установленную версию Chrome
        ChromeDriverResolver.resolve();
        driver = new ChromeDriver();
    }

    @AfterEach
    void tearDownBrowser() {
        if (driver != null) {
            driver.quit();
        }
    }

    @Step("Вход тестовым пользователем")
    protected CatalogPage loginAsTestUser() {
        return new LoginPage(driver)

package ru.qa.course.pages;

import org.openqa.selenium.By;
import org.openqa.selenium.ElementClickInterceptedException;
import org.openqa.selenium.StaleElementReferenceException;
import org.openqa.selenium.TimeoutException;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.FluentWait;
import org.openqa.selenium.support.ui.WebDriverWait;
import ru.qa.course.core.Config;

/**
 * Общие приёмы работы со страницей: все действия и чтения — через явные ожидания.
 *
 * - Элемент каждый раз ищется заново по локатору (By), ссылки на WebElement
 *   между действиями не хранятся — StaleElementReferenceException невозможен
 *   по построению, а если элемент перерисуется прямо во время действия,
 *   ожидание просто повторит попытку.
 * - Ни одного Thread.sleep: ожидание длится ровно до нужного состояния.
 */
public abstract class BasePage {

    protected final WebDriver driver;
    protected final FluentWait<WebDriver> wait;

    protected BasePage(WebDriver driver) {
        this.driver = driver;
        // WebDriverWait по умолчанию игнорирует NoSuchElementException;
        // добавляем перерисовку элемента и перекрытие кликом.
        this.wait = new WebDriverWait(driver, Config.WAIT_TIMEOUT)
                .ignoring(StaleElementReferenceException.class)
                .ignoring(ElementClickInterceptedException.class);
    }

    /** Дождаться, пока элемент станет видимым и доступным, и кликнуть по нему. */
    protected void click(By locator) {
        wait.until(d -> {
            WebElement element = d.findElement(locator);
            if (!element.isDisplayed() || !element.isEnabled()) {
                return false;
            }
            element.click();
            return true;
        });
    }

    /** Дождаться видимости поля и ввести текст. */
    protected void type(By locator, String text) {
        wait.until(ExpectedConditions.visibilityOfElementLocated(locator)).sendKeys(text);
    }

    /** Текст элемента (элемент ищется заново, перерисовка не страшна). */
    protected String text(By locator) {
        return wait.until(d -> d.findElement(locator).getText());
    }

    /**
     * Custom condition «текст элемента == expected».
     * Ждёт нужного значения; если за таймаут оно не появилось — возвращает
     * фактический текст, чтобы assertEquals в тесте показал понятную разницу
     * «ожидалось / получено», а не голый TimeoutException.
     */
    protected String textOnceEquals(By locator, String expected) {
        try {
            wait.until(d -> expected.equals(d.findElement(locator).getText()));
        } catch (TimeoutException e) {

#!/usr/bin/env bash
# Прогоны семинаров 7 и 8 в GitHub Actions (ubuntu-latest, Chrome под Xvfb).
# Результаты складываются в results/ и коммитятся workflow обратно в репозиторий.
set -u

ROOT="$(pwd)"
OUT="$ROOT/results"
rm -rf "$OUT"
mkdir -p "$OUT"
SUMMARY="$OUT/summary.txt"

{
  echo "Дата прогона (UTC): $(date -u '+%Y-%m-%d %H:%M:%S')"
  echo "Runner: $(lsb_release -ds 2>/dev/null)"
  echo "Java: $(java -version 2>&1 | head -1)"
  echo "Maven: $(mvn -v 2>/dev/null | head -1)"
  echo "Chrome: $(google-chrome --version 2>/dev/null)"
  echo
} > "$SUMMARY"

# run_suite <каталог проекта> <метка> <число прогонов>
run_suite() {
  local dir="$1" label="$2" runs="$3"
  echo "== $label ($dir), прогонов: $runs" >> "$SUMMARY"
  for i in $(seq 1 "$runs"); do
    local log="$OUT/${label}_run${i}.log"
    (cd "$dir" && rm -rf target/surefire-reports && xvfb-run -a mvn -B test > "$log" 2>&1)
    local code=$?
    local line
    line=$(grep -E "Tests run: [0-9]+, Failures: [0-9]+, Errors: [0-9]+, Skipped: [0-9]+$" "$log" | tail -1)
    local status="GREEN"
    [ "$code" -ne 0 ] && status="RED"
    echo "  прогон $i: $status | $line" >> "$SUMMARY"
    # какие тесты упали и с каким исключением
    grep -E "^\[ERROR\]   [A-Za-z]|^\[ERROR\] [a-zA-Z0-9_]+\(.*\)|Exception:" "$log" \
      | grep -vE "^\[ERROR\] (Tests run|Failures|Errors|To see|Re-run|-> \[Help|Failed to execute)" \
      | head -20 | sed 's/^/      /' >> "$SUMMARY"
    mkdir -p "$OUT/${label}_surefire_run${i}"
    cp "$dir"/target/surefire-reports/*.txt "$OUT/${label}_surefire_run${i}/" 2>/dev/null
  done
  echo >> "$SUMMARY"
}

# --- Семинар 7: до и после рефакторинга -----------------------------------
run_suite seminar7-original s7_original 1
run_suite seminar7 s7_refactored 3

# --- Семинар 8: исходный flaky-набор -----------------------------------------
run_suite seminar8-original s8_flaky_original 5

# --- Семинар 8: исправленный набор, 5 прогонов подряд + Allure с историей ---
echo "== s8_fixed (seminar8) + Allure, прогонов: 5" >> "$SUMMARY"
HIST=/tmp/allure-history
rm -rf "$HIST"
for i in 1 2 3 4 5; do
  log="$OUT/s8_fixed_run${i}.log"
  (
    cd seminar8 || exit 1
    rm -rf target/allure-results target/surefire-reports
    mkdir -p target/allure-results
    [ -d "$HIST" ] && cp -r "$HIST" target/allure-results/history
    cat > target/allure-results/executor.json <<EOF
{"name":"GitHub Actions","type":"github","buildName":"Прогон $i из 5","buildOrder":$i,
 "buildUrl":"${GITHUB_SERVER_URL:-}/${GITHUB_REPOSITORY:-}/actions/runs/${GITHUB_RUN_ID:-}"}
EOF
    {
      echo "Browser=$(google-chrome --version 2>/dev/null)"
      echo "Java=$(java -version 2>&1 | head -1 | tr -d '\"')"
      echo "OS=$(lsb_release -ds 2>/dev/null)"
      echo "App=test-app/shop.html (задержки 0-500 мс, перемешивание книг)"
    } > target/allure-results/environment.properties
    xvfb-run -a mvn -B test > "$log" 2>&1
  )
  code=$?
  line=$(grep -E "Tests run: [0-9]+, Failures: [0-9]+, Errors: [0-9]+, Skipped: [0-9]+$" "$log" | tail -1)
  status="GREEN"; [ "$code" -ne 0 ] && status="RED"
  echo "  прогон $i: $status | $line" >> "$SUMMARY"
  mkdir -p "$OUT/s8_fixed_surefire_run${i}"
  cp seminar8/target/surefire-reports/*.txt "$OUT/s8_fixed_surefire_run${i}/" 2>/dev/null
  (cd seminar8 && mvn -B allure:report > "$OUT/s8_fixed_allure_report_run${i}.log" 2>&1)
  rm -rf "$HIST"
  cp -r seminar8/target/site/allure-maven-plugin/history "$HIST" 2>/dev/null
done
cp -r seminar8/target/site/allure-maven-plugin "$OUT/allure-report"
cp -r seminar8/target/allure-results "$OUT/allure-results"
echo >> "$SUMMARY"

# --- Демонстрация скриншота при падении (копия seminar8 с намеренно неверным итогом) ---
echo "== Демонстрация: скриншот при падении в Allure (намеренно неверный ожидаемый итог)" >> "$SUMMARY"
DEMO=/tmp/seminar8-failure-demo
rm -rf "$DEMO"; cp -r seminar8 "$DEMO"
rm -rf "$DEMO/target"
sed -i 's/String.valueOf(2 \* Book.REFACTORING.price())/String.valueOf(3 * Book.REFACTORING.price())/' \
  "$DEMO/src/test/java/ru/qa/course/tests/BookShopRegressionTest.java"
(cd "$DEMO" && xvfb-run -a mvn -B test > "$OUT/s8_failure_demo.log" 2>&1)
line=$(grep -E "Tests run: [0-9]+, Failures: [0-9]+, Errors: [0-9]+, Skipped: [0-9]+$" "$OUT/s8_failure_demo.log" | tail -1)
echo "  $line" >> "$SUMMARY"
(cd "$DEMO" && mvn -B allure:report > /dev/null 2>&1)
cp -r "$DEMO/target/site/allure-maven-plugin" "$OUT/allure-report-failure-demo"

cat "$SUMMARY"
exit 0
